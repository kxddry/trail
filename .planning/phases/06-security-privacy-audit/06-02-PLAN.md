type Logger = {
  debug: (message: string, payload?: LogPayload) => void;
  info: (message: string, payload?: LogPayload) => void;
  warn: (message: string, payload?: LogPayload) => void;
  error: (message: string, payload?: LogPayload) => void;

export function createLogger(namespace: string): Logger {
  const prefix = `[${namespace}]`;
  return {
    debug(message, payload?) { if (IS_DEV) console.debug(prefix, message, payload ?? ""); },
    info(message, payload?) { if (IS_DEV) console.info(prefix, message, payload ?? ""); },
    warn(message, payload?) { console.warn(prefix, message, payload ?? ""); },
    error(message, payload?) { console.error(prefix, message, payload ?? ""); },
  };
}

    2. Add redactSensitive function:
    ```typescript
    export function redactSensitive(payload: LogPayload): LogPayload {
      const result: LogPayload = {};
      for (const [key, value] of Object.entries(payload)) {
        const lowerKey = key.toLowerCase();
        if (SENSITIVE_KEYS.has(lowerKey) || SENSITIVE_KEYS.has(key)) {
          result[key] = "[redacted]";
        } else if (URL_KEYS.has(lowerKey) || URL_KEYS.has(key)) {
          result[key] = typeof value === "string" ? domainOnly(value) : value;
        } else if (TITLE_KEYS.has(lowerKey) || TITLE_KEYS.has(key)) {
          result[key] = "[title omitted]";
        } else if (typeof value === "object" && value !== null && !Array.isArray(value)) {
          result[key] = redactSensitive(value as LogPayload);
        } else {
          result[key] = value;
        }
      }
      return result;
    }

    function domainOnly(url: string): string {
      try {
        const parsed = new URL(url);
        return parsed.hostname;
      } catch {
        return "[invalid-url]";
      }
    }
    ```

    3. Update warn and error in createLogger to auto-redact payloads:
    ```typescript
    warn(message: string, payload?: LogPayload) {
      console.warn(prefix, message, payload ? redactSensitive(payload) : "");
    },
    error(message: string, payload?: LogPayload) {
      console.error(prefix, message, payload ? redactSensitive(payload) : "");
    },
    ```

    debug and info stay as-is (dev-only, no redaction needed).

    Key design decisions:
    - Field-name matching (not regex on values) for predictability
    - Case-insensitive key check via lowercased comparison
    - One level of recursive object redaction (not deep -- payloads are flat records)
    - domainOnly extracts hostname via URL constructor, falls back to "[invalid-url]"
    - Export redactSensitive for testing and manual use
