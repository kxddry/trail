**In graph-snapshot.ts**, add a `diffSnapshots` function below `createGraphSnapshot`:

```typescript
export type SnapshotDiff = {
  readonly added: readonly NodeRenderModel[];
  readonly removed: readonly NodeId[];
  readonly moved: readonly NodeRenderModel[];
};

export function diffSnapshots(
  prev: GraphSnapshot,
  next: GraphSnapshot,
): SnapshotDiff {
  const prevById = new Map<NodeId, NodeRenderModel>();
  for (const node of prev.nodes) prevById.set(node.id, node);

  const nextById = new Map<NodeId, NodeRenderModel>();
  for (const node of next.nodes) nextById.set(node.id, node);

  const added: NodeRenderModel[] = [];
  const moved: NodeRenderModel[] = [];
  const removed: NodeId[] = [];

  for (const node of next.nodes) {
    const old = prevById.get(node.id);
    if (!old) {
      added.push(node);
    } else if (old.x !== node.x || old.y !== node.y) {
      moved.push(node);
    }
    // Nodes at same position with same id are unchanged — skip
  }

  for (const node of prev.nodes) {
    if (!nextById.has(node.id)) {
      removed.push(node.id);
    }
  }

  return { added, removed, moved };
}
```

**In GraphPixiRenderer.ts**, add `patchStaticGraph` method immediately after `setStaticGraph`:

```typescript
patchStaticGraph(snapshot: GraphSnapshot) {
  const start = performance.now();
  const prevSnapshot = this.data as GraphSnapshot;
  const diff = diffSnapshots(
    { ...prevSnapshot, nodes: Array.from(this.nodeData.values()) },
    snapshot,
  );

  // Update stored data
  this.data = { ...snapshot, particles: this.particles };
  this.connections = [...snapshot.connections];
  this.connectionsDirty = true;
  this.lastConnectionViewRect = null;

  // Patch nodeData and SpatialIndex incrementally
  for (const id of diff.removed) {
    this.nodeData.delete(id);
    this.index.remove(id);
    const visual = this.nodeMap.get(id);
    if (visual) {
      visual.destroy();
      this.nodeMap.delete(id);
    }
  }

  for (const node of diff.added) {
    this.nodeData.set(node.id, node);
    this.index.insert(node.id, {
      minX: node.x,
      minY: node.y,
      maxX: node.x + node.width,
      maxY: node.y + node.height,
    });
  }

  for (const node of diff.moved) {
    this.nodeData.set(node.id, node);
    this.index.update(node.id, {
      minX: node.x,
      minY: node.y,
      maxX: node.x + node.width,
      maxY: node.y + node.height,
    });
  }

  // Targeted invalidation (skip full invalidateAll)
  this.cachedCullingRect = null;
  this.needsRender = true;
  this.updateVisible();
  this.render();

  graphProfiler.record('patchStaticGraph', performance.now() - start);
}
```

// Static graph -- fires only when topology/layout changes (snapshot recreated on staticVersion increment)
useEffect(() => {
  const renderer = rendererRef.current;
  if (!renderer) return;
  const prevVersion = prevSnapshotVersionRef.current;
  prevSnapshotVersionRef.current = snapshot.version;

  // Use incremental patch when:
  // 1. We have a previous snapshot (not first render)
  // 2. Layout mode hasn't changed

  if (
    prevVersion > 0 &&
    snapshot.layoutMode === renderer.data.layoutMode &&
    snapshot.nodeWidth === renderer.data.nodeWidth &&
    snapshot.nodeHeight === renderer.data.nodeHeight

    renderer.patchStaticGraph(snapshot);
  } else {
    renderer.setStaticGraph(snapshot);
  }
