<objective>
Split the graph bootstrap sequence so that the first paint only waits on settings + tree state (critical path), while tag definitions, trees list, command state, and platform metadata load after first paint as deferred fetches. Add pre-mount cache fetching to sidepanel and fullview entrypoints.

Purpose: Currently `useBootstrap` awaits ALL fetches (settings, flags, tree, tags, trees list, platform info) in a single `Promise.all` before calling `bootstrap()`. This means the graph cannot render until the slowest fetch completes. By splitting into critical (settings + tree) and deferred (everything else), the graph can paint immediately from cache or the critical tree data, while non-visual data streams in afterward.

Output: New `bootstrapGraph` store action, split `useBootstrap` with deferred phase, pre-mount cache in panel bootstrap, and resilient fallback on fetch failure.
</objective>

       ```typescript
       export function useBootstrap(): void {
         const bootstrapGraph = useStore((state) => state.bootstrapGraph);
         const setPlatformMeta = useStore((state) => state.setPlatformMeta);
         const setCommandState = useStore((state) => state.setCommandState);
         const setEditorMode = useStore((state) => state.setEditorMode);
         const setSettings = useStore((state) => state.setSettings);
         const setFlags = useStore((state) => state.setFlags);
         const setTagDefs = useStore((state) => state.setTagDefs);
         const setTrees = useStore((state) => state.setTrees);

         useEffect(() => {
           let mounted = true;

           (async () => {
             // ---- Critical path: settings + tree only ----
             try {
               const [settings, treeState] = await Promise.all([
                 getSettings(),
                 getTreeState(),
               ]);
               if (!mounted) return;

               bootstrapGraph(settings, treeState);
               setSettings(settings);
               applyTheme(settings.theme, settings.customTheme);

               // Restore editor mode if active before close
               if (treeState.editorMode?.active) {
                 setEditorMode(treeState.editorMode);
               }
             } catch (error) {
               console.error('[bootstrap] Critical fetch failed, falling back to mount-then-fetch:', error);
               // Surface already mounted React tree -- it will show loading state
               // until TREE_UPDATED arrives from background and applyTreeState fires
               return;
             }

             // ---- Deferred: non-blocking fetches after first paint ----
             if (!mounted) return;

             // Fire all deferred fetches in parallel, each with independent error handling
             const deferredPromises = [
               getFlags().then((flags) => { if (mounted) setFlags(flags); }).catch((e) => console.warn('[bootstrap] flags fetch failed:', e)),
               getTagDefs().then((defs) => { if (mounted) setTagDefs(defs); }).catch((e) => console.warn('[bootstrap] tagDefs fetch failed:', e)),
               getTrees().then((snapshot) => { if (mounted) setTrees(snapshot); }).catch((e) => console.warn('[bootstrap] trees fetch failed:', e)),
               isMac().then((isMacValue) => {
                 if (!mounted) return;
                 const metaKey: MetaKeyPair = isMacValue
                   ? { short: '\u2318', long: 'cmd' }
                   : { short: '^', long: 'ctrl' };
                 setPlatformMeta(metaKey, isMacValue);
               }).catch((e) => console.warn('[bootstrap] platform meta failed:', e)),
               getCommandState().then((state) => { if (mounted) setCommandState(state); }).catch((e) => console.warn('[bootstrap] commandState fetch failed:', e)),
             ];
             await Promise.allSettled(deferredPromises);
           })();

           return () => { mounted = false; };
         }, [bootstrapGraph, setPlatformMeta, setCommandState, setEditorMode, setFlags, setSettings, setTagDefs, setTrees]);
       }
       ```

    2. Keep the same imports, add `import { isMac } from '@shared/utils/platform.js';` (already imported), add `import { MetaKeyPair } from '../state/store.js';` (already imported).

      var __trailCachedLayout: CachedLayoutResult | null | undefined;

    export type MountPanelAppOptions = {
      rootId?: string;
      connectPortName?: string;
      cachedLayout?: CachedLayoutResult | null;
    };

    export function mountPanelApp(options: MountPanelAppOptions = {}): void {
      const { rootId = 'root', connectPortName, cachedLayout } = options;

      if (connectPortName) {
        browser.runtime.connect({ name: connectPortName });
      }

      // Stash pre-fetched layout for GraphCanvas consumption
      if (cachedLayout !== undefined) {
        globalThis.__trailCachedLayout = cachedLayout;
      }

      const rootEl = document.getElementById(rootId);
      if (!rootEl) {
        throw new Error(`Missing #${rootId} element`);
      }

      void initBackendConfig();

      createRoot(rootEl).render(
        <ErrorBoundary>
          <QueryClientProvider client={queryClient}>
            <App />
          </QueryClientProvider>
        </ErrorBoundary>,
      );
    }
