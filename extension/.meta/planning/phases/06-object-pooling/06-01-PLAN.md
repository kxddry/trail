Create a `GraphicsPool` class above `GraphPixiRenderer` (after the `NodeVisual` class ends at line 727). This is a simple acquire/release pool for PIXI.Graphics objects:

```typescript
class GraphicsPool {
  private readonly pool: PIXI.Graphics[] = [];
  private activeCount = 0;

  constructor(private readonly capacity: number) {
    for (let i = 0; i < capacity; i++) {
      this.pool.push(new PIXI.Graphics());
    }
  }

  acquire(): PIXI.Graphics {
    if (this.activeCount < this.pool.length) {
      const g = this.pool[this.activeCount];
      this.activeCount++;
      g.visible = true;
      return g;
    }
    // Overflow: create new (should be rare if capacity sized correctly)
    const g = new PIXI.Graphics();
    this.pool.push(g);
    this.activeCount++;
    return g;
  }

  releaseAll() {
    for (let i = 0; i < this.activeCount; i++) {
      const g = this.pool[i];
      g.clear();
      g.visible = false;
    }
    this.activeCount = 0;
  }

  destroyAll() {
    for (const g of this.pool) {
      g.destroy();
    }
    this.pool.length = 0;
    this.activeCount = 0;
  }

  get active(): number {
    return this.activeCount;
  }
}
```

     private readonly pool: PIXI.Graphics[] = [];
     private activeCount = 0;
     private readonly parent: PIXI.Container;

     constructor(capacity: number, parent: PIXI.Container) {
       this.parent = parent;
       for (let i = 0; i < capacity; i++) {
         const g = new PIXI.Graphics();
         g.visible = false;
         parent.addChild(g);
         this.pool.push(g);
       }
     }

     acquire(): PIXI.Graphics {
       let g: PIXI.Graphics;
       if (this.activeCount < this.pool.length) {
         g = this.pool[this.activeCount];
       } else {
         g = new PIXI.Graphics();
         this.parent.addChild(g);
         this.pool.push(g);
       }
       this.activeCount++;
       g.visible = true;
       return g;
     }

     releaseAll() {
       for (let i = 0; i < this.activeCount; i++) {
         const g = this.pool[i];
         g.clear();
         g.visible = false;
       }
       this.activeCount = 0;
     }

     destroyAll() {
       for (const g of this.pool) {
         g.destroy();
       }
       this.pool.length = 0;
       this.activeCount = 0;
     }
   }
