<objective>
Add dirty flag to `updateConnections()` in GraphPixiRenderer so it skips the expensive clear+redraw cycle when connections have not changed.

Purpose: Even with Plan 01's caching, `updateConnections()` is called on every `invalidateVisible()` (animation ticks, pan, zoom). Currently it calls `connectionLayer.clear()` + `pathLayer.clear()` and redraws all visible segments every time. With the connection cache from Plan 01, the `connections` array reference stays stable during animation. A dirty flag lets the renderer skip the entire redraw loop when connections haven't changed and the viewport hasn't shifted enough to change which connections are visible.

Output: GraphPixiRenderer with dirty flag that eliminates redundant connection clears and redraws during animation, achieving >80% reduction in connection rendering time.
</objective>

```typescript
private updateConnections(viewRect: {
  minX: number; minY: number; maxX: number; maxY: number;
}): { connectionsMs: number; visibleConnections: number } {
  const start = performance.now();

  // Particles always need updating (they move every frame)
  for (const child of this.particleLayer.removeChildren()) child.destroy();
  this.updateParticles();

  if (!this.data.showConnections) {
    if (this.connectionsDirty) {
      this.connectionLayer.clear();
      this.pathLayer.clear();
      this.connectionsDirty = false;
    }
    graphProfiler.record('updateConnections', performance.now() - start);
    return { connectionsMs: performance.now() - start, visibleConnections: 0 };
  }

  const representativePx = Math.min(this.data.nodeWidth, this.data.nodeHeight) * this.view.scale;
  if (representativePx < LOD_CONN_HIDE_PX) {
    if (this.connectionsDirty) {
      this.connectionLayer.clear();
      this.pathLayer.clear();
      this.connectionsDirty = false;
    }
    graphProfiler.record('updateConnections', performance.now() - start);
    return { connectionsMs: performance.now() - start, visibleConnections: 0 };
  }

  // Skip full redraw if connections haven't changed and viewport is same
  const viewUnchanged = this.lastConnectionViewRect !== null &&
    this.lastConnectionViewRect.minX === viewRect.minX &&
    this.lastConnectionViewRect.minY === viewRect.minY &&
    this.lastConnectionViewRect.maxX === viewRect.maxX &&
    this.lastConnectionViewRect.maxY === viewRect.maxY;

  if (!this.connectionsDirty && viewUnchanged) {
    graphProfiler.record('updateConnections', performance.now() - start);
    return { connectionsMs: performance.now() - start, visibleConnections: this.metrics.visibleConnections };
  }

  // Full redraw needed
  this.connectionLayer.clear();
  this.pathLayer.clear();
  this.connectionsDirty = false;
  this.lastConnectionViewRect = { ...viewRect };

  // ... rest of existing drawing code unchanged from here (baseColor, pathColor, forEach loop, stroke) ...
  // BUT remove the this.updateParticles() call at the end (moved to top)
```
